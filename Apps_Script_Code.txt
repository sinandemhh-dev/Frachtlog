// ohmlog Backend (Google Apps Script) - schreibt Einträge in die Tabelle (erstes Blatt)
const SECRET = "frachtlog-n7q4x2"; // muss identisch mit SECRET in index.html sein
const HEAD = ["Zeitstempel","Datum","Tag","Fahrzeug","Fahrer","Auftraggeber","Fahrt-Nr","Von","Nach","Größe","Container 1","Typ 1","Container 2","Typ 2","L/V","Chassis","Auftragsnummer","Bemerkungen / Wartezeiten","ID","Chassis A-Nummer"];
const ID_COL = 19;             // Spalte S = ID
const DATA_COLS = HEAD.length; // A bis T

// Letzte Zeile mit Daten: nur Spalte A zählt
function lastDataRow(sh) {
  const a = sh.getRange("A1:A" + sh.getMaxRows()).getValues();
  let n = a.length;
  while (n > 0 && a[n - 1][0] === "") n--;
  return n;
}

function doPost(e) {
  const lock = LockService.getScriptLock();
  lock.waitLock(20000);
  try {
    const d = JSON.parse(e.postData.contents);
    if (d.secret !== SECRET) return out({ error: "unauthorized" });
    const sh = SpreadsheetApp.getActiveSpreadsheet().getSheets()[0];
    let last = lastDataRow(sh);
    if (last === 0) {
      sh.getRange(1, 1, 1, DATA_COLS).setValues([HEAD]).setFontWeight("bold");
      sh.setFrozenRows(1);
      last = 1;
    } else if (sh.getRange(1, DATA_COLS).getValue() === "") {
      // vorhandene Tabelle: Überschrift der neuen Spalte T einmalig ergänzen
      sh.getRange(1, DATA_COLS).setValue(HEAD[DATA_COLS - 1]).setFontWeight("bold");
    }
    if (last > 1) {
      const ids = sh.getRange(2, ID_COL, last - 1, 1).getValues().flat();
      if (ids.indexOf(d.id) !== -1) return out({ status: "dup" });
    }
    if (last + 1 > sh.getMaxRows()) sh.insertRowsAfter(sh.getMaxRows(), 100);
    sh.getRange(last + 1, 1, 1, DATA_COLS).setValues([[new Date(), d.datum, d.tag, d.fahrzeug, d.fahrer, d.auftraggeber, d.fahrtnr, d.von, d.nach,
      d.groesse, d.container1, d.typ1, d.container2, d.typ2, d.lv, d.chassis, d.auftragsnummer, d.bemerkung, d.id, d.chassisANummer || ""]]);
    return out({ status: "ok" });
  } catch (err) {
    return out({ error: String(err) });
  } finally {
    lock.releaseLock();
  }
}
function out(o) { return ContentService.createTextOutput(JSON.stringify(o)).setMimeType(ContentService.MimeType.JSON); }
